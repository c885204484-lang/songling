cmake_minimum_required(VERSION 3.10)
project(offline_native_regression)
find_package(ament_cmake REQUIRED)
find_package(nav2_costmap_2d REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(behaviortree_cpp_v3 REQUIRED)
add_executable(scan_clearing ../integration_offline_scan_clearing.cpp)
ament_target_dependencies(scan_clearing nav2_costmap_2d sensor_msgs)
add_executable(controller_recovery ../integration_offline_controller_recovery.cpp)
ament_target_dependencies(controller_recovery behaviortree_cpp_v3)
enable_testing()
add_executable(footprint_clearance ../integration_offline_footprint_clearance.cpp)
ament_target_dependencies(footprint_clearance nav2_costmap_2d)
add_test(NAME footprint_clearance COMMAND footprint_clearance)
add_test(NAME controller_recovery COMMAND controller_recovery
  ${CMAKE_CURRENT_SOURCE_DIR}/../../behavior_trees/offline_to_pose.xml
  ${CMAKE_CURRENT_SOURCE_DIR}/../../behavior_trees/offline_through_poses.xml
  ${CMAKE_CURRENT_SOURCE_DIR}/../../behavior_trees/offline_fixed_to_pose.xml
  ${CMAKE_CURRENT_SOURCE_DIR}/../../behavior_trees/offline_fixed_through_poses.xml)
add_test(NAME scan_clearing COMMAND scan_clearing)
set_tests_properties(scan_clearing PROPERTIES ENVIRONMENT "ROS_DOMAIN_ID=94;ROS_LOG_DIR=/tmp/songling-native-regression-ros")
find_package(rclcpp REQUIRED)
find_package(nav2_msgs REQUIRED)
add_executable(service_timeout ../integration_offline_service_timeout.cpp)
ament_target_dependencies(service_timeout behaviortree_cpp_v3 rclcpp nav2_msgs)
add_test(NAME service_timeout COMMAND service_timeout)
set_tests_properties(service_timeout PROPERTIES ENVIRONMENT "ROS_DOMAIN_ID=94;ROS_LOG_DIR=/tmp/songling-native-regression-ros")
